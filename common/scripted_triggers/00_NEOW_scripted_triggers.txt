

neow_barony_is_valid_for_city_trigger = {
	title_province = {
		has_ongoing_construction = no
		OR = {
			is_county_capital = no
			barony.holder.top_liege = { government_has_flag = government_is_republic }
		}
		has_holding_type = village_holding
	}
	trigger_if = {
		limit = {
			holder = $CHARACTER$
		}
	}
	trigger_else = {
		holder.primary_title.tier = tier_barony
	}
}
